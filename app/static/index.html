"""History utilities for recent generated audio files.""" from __future__ import annotations import json from pathlib import Path from typing import Any from uuid import uuid4 class HistoryManager: def __init__(self, history_path: Path, max_items: int = 12): self.history_path = Path(history_path) self.max_items = max_items self.history_path.parent.mkdir(parents=True, exist_ok=True) self.history_path.touch(exist_ok=True) def _load(self) -> list[dict[str, Any]]: if self.history_path.stat().st_size == 0: return [] try: with self.history_path.open("r", encoding="utf-8") as fh: content = json.load(fh) except json.JSONDecodeError: return [] if not isinstance(content, list): return [] return [item for item in content if isinstance(item, dict)] def _save(self, entries: list[dict[str, Any]]) -> None: with self.history_path.open("w", encoding="utf-8") as fh: json.dump(entries, fh, ensure_ascii=False, indent=2) def list_recent(self) -> list[dict[str, Any]]: entries = self._load() valid: list[dict[str, Any]] = [] for item in entries: path = item.get("output_path") if isinstance(path, str) and Path(path).exists(): valid.append(item) self._save(valid[: self.max_items]) return valid[: self.max_items] def record_generation( self, output_path: Path, *, mode: str, text: str, ref_audio: str | None = None, instruct: str | None = None, language: str | None = None, ) -> dict[str, Any]: entry = { "id": str(uuid4()), "timestamp": __import__("datetime").datetime.now().isoformat(timespec="seconds"), "output_path": str(output_path), "mode": mode, "text": text[:200], "ref_audio": ref_audio, "instruct": instruct, "language": language, } history = self._load() history.insert(0, entry) unique: list[dict[str, Any]] = [] seen: set[str] = set() for item in history: item_id = item.get("id") if item_id and item_id not in seen: seen.add(item_id) unique.append(item) if len(unique) >= self.max_items: break self._save(unique) return entry def delete_entry(self, entry_id: str) -> None: entries = [item for item in self._load() if item.get("id") != entry_id] self._save(entries)
