  <script>
    const statusEl = document.getElementById('status');
    const historyEl = document.getElementById('history');
    const modeEl = document.getElementById('mode');
    const cloneFields = document.getElementById('clone-fields');
    const designFields = document.getElementById('design-fields');
    const player = document.getElementById('player');
    const downloadLink = document.getElementById('download-link');

    function setStatus(message) {
      statusEl.textContent = message;
    }

    function updateModeVisibility() {
      const mode = modeEl.value;
      cloneFields.classList.toggle('hidden', mode !== 'clone');
      designFields.classList.toggle('hidden', mode !== 'design');
    }

    modeEl.addEventListener('change', updateModeVisibility);

    async function request(url, options = {}) {
      const response = await fetch(url, options);
      let data = {};
      try {
        data = await response.json();
      } catch {
        data = {};
      }
      if (!response.ok) {
        throw new Error(data.detail || `Request failed (${response.status})`);
      }
      return data;
    }

    document.getElementById('load-model').addEventListener('click', async () => {
      const device = document.getElementById('device').value;
      setStatus('Loading model...');
      try {
        const result = await request(`/api/load-model?device=${encodeURIComponent(device)}`, { method: 'POST' });
        setStatus(`${result.message} Device: ${result.device || 'unknown'}`);
      } catch (error) {
        setStatus(`Load failed: ${error.message}`);
      }
    });

    document.getElementById('generate').addEventListener('click', async () => {
      const form = new FormData();
      form.append('text', document.getElementById('text').value);
      form.append('mode', modeEl.value);
      form.append('language', document.getElementById('language').value);
      form.append('speed', document.getElementById('speed').value);
      form.append('num_step', document.getElementById('steps').value);
      form.append('ref_text', document.getElementById('ref_text').value);
      form.append('instruct', document.getElementById('instruct').value);
      form.append('consent', document.getElementById('consent').checked ? 'true' : 'false');
      const refAudio = document.getElementById('ref_audio').files[0];
      if (refAudio) form.append('ref_audio', refAudio);

      try {
        setStatus('Generating audio...');
        const result = await request('/api/generate', { method: 'POST', body: form });
        player.src = result.audio_url;
        player.load();
        document.getElementById('output-info').textContent = `${result.filename} · ${result.duration_seconds.toFixed(1)} sec`;
        downloadLink.href = result.audio_url;
        downloadLink.classList.remove('hidden');
        setStatus('Audio generated successfully.');
        await refreshHistory();
      } catch (error) {
        setStatus(`Generation failed: ${error.message}`);
      }
    });

    async function refreshHistory() {
      try {
        const items = await request('/api/history');
        historyEl.innerHTML = '';
        if (!items.length) {
          historyEl.textContent = 'No recent outputs.';
          return;
        }

        for (const item of items) {
          const row = document.createElement('div');
          row.className = 'history-item';
          row.innerHTML = `
            <strong>${item.filename}</strong>
            <div class="muted">${item.timestamp || ''} · ${item.mode || 'auto'}</div>
            <div class="muted">${(item.text || '').slice(0, 120)}</div>
            <div class="small-actions">
              <button class="play-btn" type="button">Play</button>
              <button class="delete-btn" type="button">Delete</button>
            </div>
          `;
          row.querySelector('.play-btn').addEventListener('click', () => {
            player.src = item.audio_url;
            player.play();
            document.getElementById('output-info').textContent = item.filename;
            downloadLink.href = item.audio_url;
            downloadLink.classList.remove('hidden');
          });
          row.querySelector('.delete-btn').addEventListener('click', async () => {
            try {
              await request(`/api/history/${item.id}`, { method: 'DELETE' });
              await refreshHistory();
            } catch (error) {
              setStatus(`Delete failed: ${error.message}`);
            }
          });
          historyEl.appendChild(row);
        }
      } catch (error) {
        historyEl.textContent = 'History unavailable.';
      }
    }

    updateModeVisibility();
    refreshHistory();
  </script>
